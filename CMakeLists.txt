cmake_minimum_required(VERSION 3.20)
project(fsw VERSION 0.1.0 LANGUAGES CXX)

add_executable(fsw src/main.cpp src/frontend.cpp src/check.cpp src/wasm.cpp)
target_compile_features(fsw PRIVATE cxx_std_17)
set_target_properties(fsw PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(fsw PRIVATE /W4 /permissive- /utf-8)
    set_property(TARGET fsw PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(fsw PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    find_program(NODE_EXECUTABLE node REQUIRED)
    add_test(NAME fsw-e2e COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test.mjs" "$<TARGET_FILE:fsw>")
    set_tests_properties(fsw-e2e PROPERTIES TIMEOUT 120)
endif()
install(TARGETS fsw RUNTIME DESTINATION bin)
install(FILES LICENSE DESTINATION share/licenses/fsw)
